# The files the sandbox helper runs for "ffmpeg" and "ffprobe". Built in, never read at run time:
# the helper takes a program's name from its caller and nothing else, so no argument, variable
# or setting can make it run another file (docs/adr/0089). An image with ffmpeg elsewhere is
# built with these set. Absolute, in normal form, and plain text a C++ string literal can hold.
set(ULW_SANDBOX_FFMPEG "/usr/bin/ffmpeg" CACHE STRING
    "The ffmpeg the sandbox helper runs: an absolute path, fixed at build time")
set(ULW_SANDBOX_FFPROBE "/usr/bin/ffprobe" CACHE STRING
    "The ffprobe the sandbox helper runs: an absolute path, fixed at build time")
foreach(variable ULW_SANDBOX_FFMPEG ULW_SANDBOX_FFPROBE)
    set(path "${${variable}}")
    cmake_path(NORMAL_PATH path OUTPUT_VARIABLE normal)
    if(NOT IS_ABSOLUTE "${path}" OR NOT path STREQUAL normal OR path MATCHES "/$|//|[\"\\$<>]")
        message(FATAL_ERROR "${variable} must be an absolute path in normal form, without "
            "quotes, backslashes or generator expressions: \"${path}\"")
    endif()
endforeach()

# Writes <dir>/sandbox_programs.hpp, the table of programs one build of the helper runs: pairs
# of a name and the absolute path run for it. A path may hold generator expressions. Only the
# production helper's table is checked above; the test helper's (tests/CMakeLists.txt) also
# holds paths that are wrong on purpose, for the check the helper makes of every path it runs.
function(ulw_sandbox_programs dir)
    set(pairs ${ARGN})
    list(LENGTH pairs count)
    math(EXPR odd "${count} % 2")
    if(count EQUAL 0 OR odd)
        message(FATAL_ERROR "ulw_sandbox_programs: expected name and path pairs")
    endif()
    set(entries "")
    while(pairs)
        list(POP_FRONT pairs name path)
        if(NOT name MATCHES "^[a-z][a-z0-9_]*$")
            message(FATAL_ERROR "ulw_sandbox_programs: not a program name: \"${name}\"")
        endif()
        string(APPEND entries "    Program{.name = \"${name}\", .path = \"${path}\"},\n")
    endwhile()
    file(GENERATE OUTPUT "${dir}/sandbox_programs.hpp" CONTENT
"// Generated by infra/ffmpeg/CMakeLists.txt (ulw_sandbox_programs); do not edit.
#pragma once

#include \"program_check.hpp\"

#include <array>

namespace infra::ffmpeg::sandbox {

inline constexpr std::array kPrograms{
${entries}};

} // namespace infra::ffmpeg::sandbox
")
endfunction()

# The production helper's table: ffmpeg and ffprobe, nothing else.
set(ULW_SANDBOX_PROGRAMS_DIR "${CMAKE_CURRENT_BINARY_DIR}/sandbox_programs" CACHE INTERNAL
    "Where the production sandbox helper's generated program table is")
ulw_sandbox_programs("${ULW_SANDBOX_PROGRAMS_DIR}"
    ffmpeg "${ULW_SANDBOX_FFMPEG}"
    ffprobe "${ULW_SANDBOX_FFPROBE}")

# Every ffmpeg and ffprobe child is started through this; see src/sandbox_main.cpp. Never
# sanitized, so it links no instrumented library either: ASan's runtime needs fresh mappings
# after the helper has lowered RLIMIT_AS below the terabytes of shadow memory it reserved, and
# dies trying. It uses only header-only parts of core. The images carry this one alone; the
# test helper (tests/CMakeLists.txt) is built from the same source with a table of its own.
add_executable(ulw_sandbox src/sandbox_main.cpp)
target_compile_features(ulw_sandbox PRIVATE cxx_std_23)
target_include_directories(ulw_sandbox PRIVATE ${ULW_SANDBOX_PROGRAMS_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/src ${PROJECT_SOURCE_DIR}/core/include)
target_link_libraries(ulw_sandbox PRIVATE ulw_warnings)
